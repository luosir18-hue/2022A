```
常量: m_f=4866, m_o=2433, m_a=1335.535, b_h=656.3616, F=6250, ω=1.4005
      ρ=1025, g=9.8, R_f=1, k=80000
      K_h = ρ·g·π·R_f²
      情形一: C = 10000
      情形二: c0 = 10000, p = 0.5
      dt = 0.001, T = 2π/ω, t_max = 40·T

函数 D(v_r):
    if 情形一: return C · v_r
    else:      return c0 · |v_r|^p · v_r      # 与 v_r 同号，保证耗散非负

函数 rhs(t, x) -> dx:
    ξ  = x[2] − x[0]        # 相对位移
    v_r = x[3] − x[1]        # 相对速度
    D  = D(v_r)
    exc = F·cos(ω·t)
    dx[0] = x[1]
    dx[1] = (exc − b_h·x[1] − K_h·x[0] + k·ξ + D) / (m_f + m_a)
    dx[2] = x[3]
    dx[3] = (−k·ξ − D) / m_o
    return dx

函数 rk4(t, x, dt):                        # 经典四阶 RK
    k1 = rhs(t, x)
    k2 = rhs(t+dt/2, x + dt/2·k1)
    k3 = rhs(t+dt/2, x + dt/2·k2)
    k4 = rhs(t+dt,   x + dt·k3)
    x  = x + dt/6·(k1 + 2k2 + 2k3 + k4)

for type in {情形一, 情形二}:
    x = (0,0,0,0); t = 0; n = 0
    记录 (t=0, x) 到 采样表
    while t < t_max:
        rk4(t, x, dt)
        t += dt; n += 1
        if n mod 200 == 0:                  # 0.2 s 采样
            记录 (t=n·0.2/200, x) 到 采样表
    写出 result1-<case>.xlsx（表头：时间/浮子{位移,速度}/振子{位移,速度}）
    抽取 t=10,20,40,60,100 s 行供论文引用
```
